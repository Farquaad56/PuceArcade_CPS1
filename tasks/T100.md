```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)] pub enum Load { Byte, Byte16, Word16, Word16Swap, Word64, Byte64, Byte32 }
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Op { Load { kind: Load, file: &'static str, offset: u32, size: u32, crc: u32 }, Continue { offset: u32, size: u32 },
              Reload { offset: u32, size: u32 }, Ignore { size: u32 }, Copy { src: &'static str, src_offset: u32, offset: u32, size: u32 },
              Fill { offset: u32, size: u32, value: u32 }, NoDump }
#[derive(Debug)] pub struct RegionDef { pub name: &'static str, pub size: u32, pub fill: u8, pub ops: &'static [Op] }
#[derive(Debug)] pub struct RomSetDef { pub name: &'static str, pub regions: &'static [RegionDef] }
#[derive(Debug)] pub struct GameInfo { pub name: &'static str, pub parent: &'static str, pub machine: &'static str, pub input: &'static str,
    pub init: &'static str, pub rotation: u16 /* 0 ou 270 (ROTxx de MAME) */, pub maker: &'static str, pub title: &'static str, pub not_working: bool, pub romset: &'static RomSetDef }
pub fn find_game(name: &str) -> Option<&'static GameInfo>;   // recherche linéaire dans romsets_data::GAMES
```
3. `source.rs` :
```rust
pub trait FileSource { fn fetch(&self, file: &str, crc: u32) -> Result<Vec<u8>, CartError>; }   // MissingFile si absent
pub struct DirSource { pub dir: PathBuf }      // <dir>/<file> ; si absent : recherche insensible à la casse dans le dossier
pub struct ZipSource { pub path: PathBuf }     // ouvre le zip à chaque fetch ; nom insensible à la casse, sinon entrée dont le CRC32 du zip == crc
pub struct ChainSource { pub sources: Vec<Box<dyn FileSource>> }   // premier succès ; sinon MissingFile
pub fn sources_for(cfg: &Config, info: &GameInfo) -> ChainSource;
// ordre : <roms>/<name>.zip, <roms>/<name>/, puis si parent non vide : <roms>/<parent>.zip, <roms>/<parent>/ (seuls ceux qui existent)
```
Erreurs d'E/S et de zip -> `CartError::Io(message)`.
