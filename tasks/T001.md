```rust
// clock.rs — 1 tick = 1/5_040_000_000 s (ADR-001). Chaque constante porte `// spec: docs/spec/00 §...`
pub type Tick = u64;
pub const TICKS_PER_SECOND: u64 = 5_040_000_000;
pub const TICKS_PER_68K_10MHZ: u64 = 504;
pub const TICKS_PER_68K_12MHZ: u64 = 420;
pub const TICKS_PER_Z80: u64 = 1408;          // 3,579545 MHz = 315/88 MHz
pub const TICKS_PER_YM: u64 = 1408;           // horloge du YM2151 (même quartz)
pub const YM_CLOCKS_PER_SAMPLE: u64 = 64;
pub const TICKS_PER_YM_SAMPLE: u64 = 90_112;
pub const TICKS_PER_OKI: u64 = 5040;          // 1 MHz
pub const OKI_DIVISOR_PIN7_HIGH: u64 = 132;
pub const OKI_DIVISOR_PIN7_LOW: u64 = 165;
pub const TICKS_PER_PIXEL: u64 = 630;         // 8 MHz
pub const PIXELS_PER_LINE: u64 = 512;
pub const LINES_PER_FRAME: u64 = 262;
pub const TICKS_PER_LINE: u64 = 322_560;
pub const TICKS_PER_FRAME: u64 = 84_510_720;
pub const VBLANK_START_LINE: u64 = 240;
pub const TICKS_VBLANK_START: u64 = 77_414_400;  // 240 * TICKS_PER_LINE, depuis le début de trame
pub const VISIBLE_X0: usize = 64;
pub const VISIBLE_Y0: usize = 16;

// video.rs
pub const SCREEN_W: usize = 384;
pub const SCREEN_H: usize = 224;
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Framebuffer { pub pixels: Vec<u32> }   // 0x00RRGGBB, SCREEN_W*SCREEN_H, ligne par ligne
impl Framebuffer { pub fn new() -> Self; pub fn crc32(&self) -> u32; pub fn write_ppm(&self, path: &std::path::Path) -> std::io::Result<()>; }
impl Default for Framebuffer { fn default() -> Self { Self::new() } }

// input.rs — bits actifs-BAS comme le matériel (1 = relâché), docs/spec/01 §Entrées
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub struct InputState { pub in0: u8, pub in1: u16, pub in2: u8, pub in3: u8 }
impl Default for InputState { fn default() -> Self { InputState { in0: 0xFF, in1: 0xFFFF, in2: 0xFF, in3: 0xFF } } }
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub struct Dips { pub dswa: u8, pub dswb: u8, pub dswc: u8 }
impl Default for Dips { fn default() -> Self { Dips { dswa: 0xFF, dswb: 0xFF, dswc: 0xFF } } }
pub trait InputProvider { fn inputs(&self) -> InputState; }
pub mod in0 { pub const COIN1: u8 = 0x01; pub const COIN2: u8 = 0x02; pub const SERVICE1: u8 = 0x04; pub const START1: u8 = 0x10; pub const START2: u8 = 0x20; pub const TEST: u8 = 0x40; }
pub mod pad { pub const RIGHT: u8 = 0x01; pub const LEFT: u8 = 0x02; pub const DOWN: u8 = 0x04; pub const UP: u8 = 0x08; pub const B1: u8 = 0x10; pub const B2: u8 = 0x20; pub const B3: u8 = 0x40; }
pub mod kick { pub const P1_B4: u8 = 0x01; pub const P1_B5: u8 = 0x02; pub const P1_B6: u8 = 0x04; pub const P2_B4: u8 = 0x10; pub const P2_B5: u8 = 0x20; pub const P2_B6: u8 = 0x40; }

// crc.rs
pub fn crc32_bytes(data: &[u8]) -> u32;   // CRC-32/ISO-HDLC (poly réfléchi 0xEDB88320, init/xorout 0xFFFFFFFF), table OnceLock<[u32;256]>

// bits.rs — COMBINE_DATA de MAME (docs/spec/01 §Conventions)
pub fn combine_data(old: u16, data: u16, uds: bool, lds: bool) -> u16;  // mask = (uds?0xFF00:0)|(lds?0x00FF:0) ; (old & !mask) | (data & mask)
```
`Framebuffer::crc32` = `crc32_bytes` sur les pixels sérialisés en u32 little-endian (4 octets/pixel, ligne par ligne). `write_ppm` : P6 binaire « 384 224 255 », R = bits 16-23, G = 8-15, B = 0-7. Pas de dépendance externe.

## Tests à écrire (valeurs exactes)
- `TICKS_PER_FRAME == TICKS_PER_LINE * LINES_PER_FRAME` ; `TICKS_PER_LINE == PIXELS_PER_LINE * TICKS_PER_PIXEL` ; `TICKS_VBLANK_START == VBLANK_START_LINE * TICKS_PER_LINE` ; `TICKS_PER_YM_SAMPLE == YM_CLOCKS_PER_SAMPLE * TICKS_PER_YM` ; `TICKS_PER_SECOND / TICKS_PER_68K_10MHZ == 10_000_000` ; `TICKS_PER_SECOND % TICKS_PER_Z80 == 640` (3 579 545,45… Hz n'est pas entier).
- `Framebuffer::new().pixels.len() == 86016`, tous nuls ; `crc32_bytes(b"123456789") == 0xCBF43926` ; `Framebuffer::new().crc32() == 0x61605322`.
- `write_ppm` dans un dossier temporaire (std::env::temp_dir) : taille = 15 + 384*224*3 = 258063 octets (en-tête `P6\n384 224\n255\n`).
- `combine_data(0x1234, 0xABCD, true, false) == 0xAB34` ; `(.., false, true) == 0x12CD` ; `(.., true, true) == 0xABCD` ; `(.., false, false) == 0x1234`.
- `InputState::default().in1 == 0xFFFF` ; `Dips::default().dswc == 0xFF`.
