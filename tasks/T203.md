pub struct VideoBases { pub scroll: [usize; 3], pub obj: usize, pub other: usize, pub scrollx: [i32; 3], pub scrolly: [i32; 3],
    pub starsx: [i32; 2], pub starsy: [i32; 2], pub layer_enabled: [bool; 3], pub stars_enabled: [bool; 2] }
pub struct Video {
    pub cfg: &'static GameConfig,          // serde : skip, défaut find_config("cps2") ; relink() après chargement
    pub game: String,
    pub regs_a: [u16; 32], pub regs_b: [u16; 32], pub gfxram: Vec<u16>, pub palette: Vec<u32>, pub buffered_obj: Vec<u16>,
    pub bases: VideoBases, pub in2: u16, pub in3: u16, pub last_sprite: i32,
    pub debug_layer_mask: u32,             // débogueur (frontal JAMMA) : bit0 sprites, bit1-3 scroll1-3, bit4 étoiles ; 0xFFFF_FFFF au reset

impl Video {

    pub fn reset(&mut self);                         // video_start l.2533-2596 : gfxram 0, regs 0, défauts OBJ 0x9200, S1 0x9000, S2 0x9040, S3 0x9080, OTHER 0x9100 ; palette noire ; buffered_obj 0 ; in2 = in3 = 0x00FF ; get_video_base()

    pub fn gfxram_write(&mut self, word: usize, data: u16, uds: bool, lds: bool);  // COMBINE_DATA (pas de cache de tuiles : rien d'autre)
    pub fn set_inputs(&mut self, in2: u8, in3: u8);  // in2 = in2 as u16 (bits 8-15 = 0), idem in3
    pub fn get_video_base(&mut self);                // port 1:1 de cps1_get_video_base l.2246-2366, y compris les branches bootleg_kludge
    pub fn vblank(&mut self);                        // screen_vblank_cps1 puis cps1_objram_latch : get_video_base(), puis buffered_obj[i] = gfxram[(obj + i) % GFXRAM_WORDS], i < 0x400

## Règles de portage C++ -> Rust (cartes « port »)
- Traduire fonction par fonction, DANS L'ORDRE du fichier source, mêmes noms en snake_case (`cps1_build_palette` -> `cps1_build_palette` ou nom indiqué). Ne pas « optimiser », ne pas fusionner, ne pas réordonner.
- Types : `int` -> i32, `uint8_t/uint16_t/uint32_t` -> u8/u16/u32 ; arithmétique non signée modulo 2^N : `wrapping_*` ou masques explicites ; division entière C++ = division Rust sur i32 (troncature vers 0).
- Conserver l'en-tête de licence (`// license:BSD-3-Clause` + `// copyright-holders:` du fichier source) en tête de chaque fichier Rust porté, avec le nom du fichier source et les lignes portées.
- Construction sans équivalent évident : écrire la plus simple équivalente et noter `// PORT-NOTE:` ; ne jamais changer le comportement.
